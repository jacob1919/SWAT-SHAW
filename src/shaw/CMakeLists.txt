cmake_minimum_required(VERSION 3.20)
project(shaw_kernel LANGUAGES Fortran)
# The upstream executable forces free form and implicit-none flags globally.
# Legacy SHAW has explicit fixed form and standard implicit typing.
set(CMAKE_Fortran_FLAGS "")
if(WIN32 AND CMAKE_Fortran_COMPILER_ID STREQUAL "IntelLLVM")
  set(CMAKE_Fortran_FLAGS "/real-size:64 /align:nocommons /align:nodcommons")
  set(CMAKE_Fortran_FLAGS_RELEASE "/O2")
elseif(CMAKE_Fortran_COMPILER_ID MATCHES "Intel")
  set(CMAKE_Fortran_FLAGS "-real-size 64 -align nocommons -align nodcommons")
else()
  set(CMAKE_Fortran_FLAGS "-fdefault-real-8 -fdefault-double-8 -fno-align-commons")
endif()
enable_testing()
add_library(shaw_kernel STATIC shaw_water_heat.for shaw_legacy_state.f90 shaw_common_access.f90 shaw_column_types.f90 shaw_column_api.f90 shaw_canopy_jacobian.f90 shaw_root_partition.f90)
set_target_properties(shaw_kernel PROPERTIES Fortran_MODULE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/mod")
target_include_directories(shaw_kernel PUBLIC "${CMAKE_CURRENT_BINARY_DIR}/mod")
if(WIN32 AND CMAKE_Fortran_COMPILER_ID MATCHES "Intel")
  target_compile_options(shaw_kernel PRIVATE /fpe:0 /traceback /check:bounds /warn:interfaces)
  set_source_files_properties(shaw_water_heat.for PROPERTIES COMPILE_OPTIONS "/fixed;/extend-source:132")
elseif(CMAKE_Fortran_COMPILER_ID MATCHES "Intel")
  target_compile_options(shaw_kernel PRIVATE -fpe0 -traceback -check bounds -warn interfaces)
  set_source_files_properties(shaw_water_heat.for PROPERTIES COMPILE_OPTIONS "-fixed;-extend-source;132")
else()
  target_compile_options(shaw_kernel PRIVATE -fcheck=bounds -ffpe-trap=invalid,zero,overflow -fbacktrace)
  set_source_files_properties(shaw_water_heat.for PROPERTIES COMPILE_OPTIONS "-ffixed-form;-ffixed-line-length-132;-fallow-argument-mismatch")
endif()
add_executable(test_shaw_columns ../../tests/test_shaw_columns.f90)
target_link_libraries(test_shaw_columns PRIVATE shaw_kernel)
add_test(NAME shaw_column_isolation COMMAND test_shaw_columns)
add_executable(test_shaw_canopy ../../tests/test_shaw_canopy.f90)
target_link_libraries(test_shaw_canopy PRIVATE shaw_kernel)
add_test(NAME shaw_dynamic_canopy COMMAND test_shaw_canopy)
add_executable(test_shaw_conductance ../../tests/test_shaw_conductance.f90)
target_link_libraries(test_shaw_conductance PRIVATE shaw_kernel)
add_test(NAME shaw_conductance_derivative COMMAND test_shaw_conductance)
add_executable(test_shaw_roots ../../tests/test_shaw_roots.f90)
target_link_libraries(test_shaw_roots PRIVATE shaw_kernel)
add_test(NAME shaw_root_supply COMMAND test_shaw_roots)
add_executable(replay_shaw_column ../../tests/replay_shaw_column.f90)
target_link_libraries(replay_shaw_column PRIVATE shaw_kernel)
add_executable(ported_shaw_reference ../../tests/test_shaw_reference.f90)
target_link_libraries(ported_shaw_reference PRIVATE shaw_kernel)
